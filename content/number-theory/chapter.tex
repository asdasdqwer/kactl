	\subsection{Bézout's identity}
	For $a \neq $, $b \neq 0$, then $d=gcd(a,b)$ is the smallest positive integer for which there are integer solutions to
	$$ax+by=d$$
	If $(x,y)$ is one solution, then all solutions are given by
	$$\left(x+\frac{kb}{\gcd(a,b)}, y-\frac{ka}{\gcd(a,b)}\right), \quad k\in\mathbb{Z}$$

\section{Pythagorean Triples}
 The Pythagorean triples are uniquely generated by
 \[ a=k\cdot (m^{2}-n^{2}),\ \,b=k\cdot (2mn),\ \,c=k\cdot (m^{2}+n^{2}), \]
 with $m > n > 0$, $k > 0$, $m \bot n$, and either $m$ or $n$ even.

	$p=962592769$ is such that $2^{21} \mid p-1$, which may be useful. For hashing

	For $p = 2, a > 2$, the group $\mathbb Z_{2^a}^\times$ is instead isomorphic to $\mathbb Z_2 \times \mathbb Z_{2^{a-2}}$.

\section{Möbius inversion}
\textbf{Posets.} $g(t) = \sum_{s \le t} f(s) \iff f(t) = \sum_{s \le t} \mu(s, t)\, g(s)$, where
$\mu(t, t) = 1$, $\mu(s, t) = -\sum_{s < u \le t} \mu(u, t)$ for $s < t$, and $0$ otherwise.
For the reversed order use $\mu(t, s)$. Subsets: $\mu(s, t) = (-1)^{|t \setminus s|}$.
Divisibility: $\mu(s, t) = \mu(t/s)$ with
\[ \mu(n) = \begin{cases} 0 & n \text{ not square-free} \\ (-1)^k & n \text{ has } k \text{ distinct prime factors} \end{cases} \]
\textbf{Number theory.} $\sum_{d \mid n} \mu(d) = [n = 1]$ (very useful), and
\begin{gather*}
g(n) = \textstyle\sum_{d \mid n} f(d) \iff f(n) = \sum_{d \mid n} \mu(d)\, g(n/d) \\
g(n) = \textstyle\sum_{n \mid d} f(d) \iff f(n) = \sum_{n \mid d} \mu(d/n)\, g(d) \\
g(n) = \textstyle\sum_{m=1}^{n} f(\lfloor n/m \rfloor) \iff f(n) = \sum_{m=1}^{n} \mu(m)\, g(\lfloor n/m \rfloor)
\end{gather*}
\textbf{Convolutions} $c_k = \sum_{i \circ j = k} a_i b_j$: sum $a$ and $b$ over the order ($\zeta$),
multiply pointwise, then invert with $\mu$. The order is supersets for AND, subsets for OR,
multiples for $\gcd$ and divisors for $\operatorname{lcm}$.
